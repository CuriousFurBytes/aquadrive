function switchDemoTab(btn, index) {
    var tabs = btn.parentElement;
    var panels = tabs.nextElementSibling;
    tabs.querySelectorAll('.tab').forEach(t => t.classList.remove('active'));
    panels.querySelectorAll('.tab-content').forEach(p => p.classList.remove('active'));
    btn.classList.add('active');
    panels.children[index].classList.add('active');
}

function groupTab(headerId, panelsId, btn, index) {
    var header = document.getElementById(headerId);
    var panels = document.getElementById(panelsId);
    header.querySelectorAll('.tab-group-btn').forEach(t => t.classList.remove('active'));
    panels.querySelectorAll('.tab-group-panel').forEach(p => p.classList.remove('active'));
    btn.classList.add('active');
    panels.children[index].classList.add('active');
}
